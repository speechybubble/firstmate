## Scenario: lease release confirmation (lab FM_HOME=/tmp/fm-lab.cSjR2k)
$ bin/fm-lease.sh claim fm-demo-task
[exit 0]

$ bin/fm-lease.sh check fm-demo-task
[stdout] main 3999503 1792148148 stale
[exit 0]

$ bin/fm-lease.sh release fm-demo-task
[stdout] lease release: released the main supervision actor's lease on 'fm-demo-task'
[exit 0]

$ bin/fm-lease.sh check fm-demo-task
[exit 1]

## Retry release (already released)

[exit 0]

$ bin/fm-lease.sh release fm-demo-task

[exit 0]

$ bin/fm-lease.sh check fm-demo-task
